def softmax(x: list[int | float]) -> list[int | float]:
    if len(x) == 0:
        raise Exception("x is empty")
    max = x[0]
    sum = 0
    for e in x:
        if e > max:
            max = e
    for e in x:
        sum += math.exp(e - max)
    result = []
    for e in x:
        result.append(math.exp(e - max) / sum)
    return result
